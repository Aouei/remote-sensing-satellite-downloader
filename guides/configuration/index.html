    Configuration¶

    This guide explains the configuration options available in the library.

    Environment Variables¶

    Credentials¶

    To avoid hardcoding credentials in code, use environment variables:

    Linux/macOS: 

    export COPERNICUS_USER="your_email@example.com"
export COPERNICUS_PASS="your_password"
export USGS_USER="your_username"
export USGS_TOKEN="your_api_token"

    Windows (PowerShell): 

    $env:COPERNICUS_USER="your_email@example.com"
$env:COPERNICUS_PASS="your_password"
$env:USGS_USER="your_username"
$env:USGS_TOKEN="your_api_token"

    In Python: 

    import os
from sat_download.api.odata import ODataAPI

api = ODataAPI(
    username=os.getenv("COPERNICUS_USER"),
    password=os.getenv("COPERNICUS_PASS")
)

    .env File¶

    Use python-dotenv to manage credentials:

    pip install python-dotenv

    Create .env: 

    COPERNICUS_USER=your_email@example.com
COPERNICUS_PASS=your_password
USGS_USER=your_username
USGS_TOKEN=your_api_token

    In your code: 

    from dotenv import load_dotenv
import os

# Load environment variables
load_dotenv()

# Use
from sat_download.api.odata import ODataAPI

api = ODataAPI(
    username=os.getenv("COPERNICUS_USER"),
    password=os.getenv("COPERNICUS_PASS")
)

    API Configuration¶

    Verbosity¶

    Control output level during downloads:

    from sat_download.services.downloader import SatelliteImageDownloader

# Silent (no progress bar)
downloader = SatelliteImageDownloader(api, verbose=0)

# With progress bar (recommended)
downloader = SatelliteImageDownloader(api, verbose=1)

    Custom Timeouts¶

    For slow connections, configure timeouts in requests:

    import requests
from sat_download.api.odata import ODataAPI

# Configure session with custom timeout
session = requests.Session()
session.timeout = 300  # 5 minutes

# Note: Currently the library does not expose this parameter
# Consider contributing this functionality

    Search Configuration¶

    Result Limits¶

    APIs have limits on the number of results per query. Use bulk_search() for large ranges:

    from sat_download.data_types.search import SearchFilters

# For small ranges
filters = SearchFilters(
    collection="SENTINEL-2",
    start_date="2024-01-01",
    end_date="2024-01-31",  # 1 month
    tile_id="30TWM"
)
results = downloader.search(filters)

# For large ranges
filters = SearchFilters(
    collection="SENTINEL-2",
    start_date="2023-01-01",
    end_date="2023-12-31",  # 1 year
    tile_id="30TWM"
)
results = downloader.bulk_search(filters)  # Automatically divides

    Default Filters¶

    Create helper functions with default values:

    from sat_download.data_types.search import SearchFilters
from sat_download.enums import COLLECTIONS

def create_s2_filters(tile_id, start_date, end_date, level="L2A"):
    """Create Sentinel-2 filters with default values."""
    return SearchFilters(
        collection=COLLECTIONS.SENTINEL_2.value,
        processing_level=level,
        start_date=start_date,
        end_date=end_date,
        tile_id=tile_id
    )

# Use
filters = create_s2_filters("30TWM", "2024-01-01", "2024-01-31")

    Download Configuration¶

    Directory Structure¶

    Organize downloads with custom structure:

    import os
from datetime import datetime

def organize_by_date(results, base_dir):
    """Organize downloads by year/month."""
    downloads = {}

    for product_id, image in results.items():
        # Extract year and month
        year = image.date[:4]
        month = image.date[4:6]

        # Create path
        outdir = os.path.join(base_dir, year, month)

        # Save mapping
        if outdir not in downloads:
            downloads[outdir] = {}
        downloads[outdir][product_id] = image

    # Download each group to its directory
    all_paths = []
    for outdir, images in downloads.items():
        print(f"\nDownloading to {outdir}...")
        paths = downloader.bulk_download(images, outdir)
        all_paths.extend(paths)

    return all_paths

# Use
results = downloader.search(filters)
paths = organize_by_date(results, "./sentinel2_archive")

# Resulting structure:
# sentinel2_archive/
#   ├── 2024/
#   │   ├── 01/
#   │   │   ├── image1.zip
#   │   │   └── image2.zip
#   │   └── 02/
#   │       └── image3.zip

    Custom Naming¶

    Rename files during download:

    import os

def download_with_custom_names(downloader, results, outdir, pattern="{date}_{tile}"):
    """Download with custom names."""
    paths = []

    os.makedirs(outdir, exist_ok=True)

    for product_id, image in results.items():
        # Generate custom name
        custom_name = pattern.format(
            date=image.date,
            tile=image.tile,
            sensor=image.sensor,
            satellite=image.identifier
        )

        # Keep original extension
        ext = os.path.splitext(image.filename)[1]
        custom_filename = f"{custom_name}{ext}"

        # Download
        path = downloader.download(product_id, outdir, custom_filename)
        paths.append(path)

    return paths

# Use
results = downloader.search(filters)
paths = download_with_custom_names(
    downloader,
    results,
    "./downloads",
    pattern="{sensor}_{date}_{tile}"
)

# Result: Sentinel-2_20240115_30TWM.zip

    Logging to File¶

    import logging

# Configure logging to file and console
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('satellite_downloads.log'),
        logging.StreamHandler()
    ]
)

logger = logging.getLogger(__name__)

# Wrapper with logging
from sat_download.services.downloader import SatelliteImageDownloader

class LoggedDownloader(SatelliteImageDownloader):
    """Downloader with additional logging."""

    def search(self, filters):
        logger.info(f"Searching: {filters.collection} from {filters.start_date} to {filters.end_date}")
        results = super().search(filters)
        logger.info(f"Found {len(results) if results else 0} images")
        return results

    def bulk_download(self, images, outdir):
        logger.info(f"Starting download of {len(images)} images to {outdir}")
        paths = super().bulk_download(images, outdir)
        successful = len([p for p in paths if p])
        logger.info(f"Download completed: {successful}/{len(images)} successful")
        return paths

# Use
api = ODataAPI(...)
downloader = LoggedDownloader(api, verbose=1)

    Proxy¶

    To use HTTP/HTTPS proxy:

os.environ['HTTP_PROXY'] = 'http://proxy.example.com:8080'
os.environ['HTTPS_PROXY'] = 'http://proxy.example.com:8080'

# Or in requests directly (requires modifying source code)
import requests

proxies = {
    'http': 'http://proxy.example.com:8080',
    'https': 'http://proxy.example.com:8080'
}

session = requests.Session()
session.proxies.update(proxies)

    Results Cache¶

    Implement caching to avoid repeated searches:

    import json
import hashlib
from pathlib import Path

class CachedDownloader:
    """Downloader with search results cache."""

    def __init__(self, api, cache_dir=".cache"):
        self.downloader = SatelliteImageDownloader(api, verbose=1)
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(exist_ok=True)

    def _get_cache_key(self, filters):
        """Generate unique key for a filter set."""
        filter_str = f"{filters.collection}_{filters.start_date}_{filters.end_date}_{filters.tile_id}"
        return hashlib.md5(filter_str.encode()).hexdigest()

    def search_cached(self, filters):
        """Search with cache."""
        cache_key = self._get_cache_key(filters)
        cache_file = self.cache_dir / f"{cache_key}.json"

        # Try loading from cache
        if cache_file.exists():
            print(f"Loading from cache: {cache_file}")
            with open(cache_file, 'r') as f:
                cached_data = json.load(f)
            # Convert back to objects (simplified)
            return cached_data

        # Search normally
        print("No cache, searching...")
        results = self.downloader.search(filters)

        # Save in cache
        if results:
            # Serialize results (simplified)
            cache_data = {
                k: {
                    'date': v.date,
                    'sensor': v.sensor,
                    'filename': v.filename,
                    'tile': v.tile
                }
                for k, v in results.items()
            }

            with open(cache_file, 'w') as f:
                json.dump(cache_data, f, indent=2)
            print(f"Saved in cache: {cache_file}")

        return results

# Use
cached = CachedDownloader(api)
results = cached.search_cached(filters)  # First time: searches
results = cached.search_cached(filters)  # Second time: uses cache

    Download Parallelization¶

    Download multiple files in parallel:

    from concurrent.futures import ThreadPoolExecutor, as_completed
import os

def parallel_download(downloader, results, outdir, max_workers=4):
    """Download images in parallel."""
    os.makedirs(outdir, exist_ok=True)

    def download_one(item):
        product_id, image = item
        return downloader.download(product_id, outdir, image.filename)

    paths = []

    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        # Submit all downloads
        future_to_image = {
            executor.submit(download_one, item): item[1]
            for item in results.items()
        }

        # Process as they complete
        for idx, future in enumerate(as_completed(future_to_image), 1):
            image = future_to_image[future]
            try:
                path = future.result()
                paths.append(path)
                status = "✓" if path else "✗"
                print(f"[{idx}/{len(results)}] {status} {image.filename}")
            except Exception as e:
                print(f"Error downloading {image.filename}: {e}")
                paths.append(None)

    return paths

# Use
results = downloader.search(filters)
paths = parallel_download(downloader, results, "./downloads", max_workers=4)

    Configuration File¶

    Create a centralized configuration file:

    config.yaml: 

    # API configuration
apis:
  copernicus:
    username: ${COPERNICUS_USER}
    password: ${COPERNICUS_PASS}
  usgs:
    username: ${USGS_USER}
    token: ${USGS_TOKEN}

# Download configuration
download:
  verbose: 1
  output_base: "./satellite_data"
  max_retries: 3
  retry_delay: 60

# Default filters
defaults:
  sentinel2:
    processing_level: "L2A"
  landsat8:
    processing_level: "T1"

# Tiles of interest
tiles:
  madrid: "30TWM"
  barcelona: "31TBF"
  seville: "30SUF"

    Load configuration: 

    import yaml
import os
from string import Template

def load_config(config_file="config.yaml"):
    """Load configuration with environment variable substitution."""
    with open(config_file, 'r') as f:
        content = f.read()

    # Substitute environment variables
    template = Template(content)
    content_with_env = template.safe_substitute(os.environ)

    config = yaml.safe_load(content_with_env)
    return config

# Use
config = load_config()

from sat_download.api.odata import ODataAPI

api = ODataAPI(
    username=config['apis']['copernicus']['username'],
    password=config['apis']['copernicus']['password']
)

downloader = SatelliteImageDownloader(
    api,
    verbose=config['download']['verbose']
)

# Use default values
filters = SearchFilters(
    collection=COLLECTIONS.SENTINEL_2.value,
    processing_level=config['defaults']['sentinel2']['processing_level'],
    start_date="2024-01-01",
    end_date="2024-01-31",
    tile_id=config['tiles']['madrid']
)
