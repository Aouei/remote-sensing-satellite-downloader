    Advanced Examples¶

    This guide provides complete examples for common and advanced use cases.

    Example 1: Temporal Analysis¶

    Download time series from a specific area:

    from sat_download.api.odata import ODataAPI
from sat_download.services.downloader import SatelliteImageDownloader
from sat_download.data_types.search import SearchFilters
from sat_download.enums import COLLECTIONS
from datetime import datetime, timedelta
import os

def download_time_series(tile_id, start_date, end_date, outdir):
    """
    Download Sentinel-2 time series for analysis.
    """
    # Configure
    api = ODataAPI(username=os.getenv("COPERNICUS_USER"), 
                   password=os.getenv("COPERNICUS_PASS"))
    downloader = SatelliteImageDownloader(api, verbose=1)

    # Search all images in range
    filters = SearchFilters(
        collection=COLLECTIONS.SENTINEL_2.value,
        processing_level="L2A",
        start_date=start_date,
        end_date=end_date,
        tile_id=tile_id
    )

    print(f"Searching time series {start_date} to {end_date}...")
    results = downloader.bulk_search(filters)

    # Group by month
    by_month = {}
    for image in results.values():
        month = image.date[:6]  # YYYYMM
        if month not in by_month:
            by_month[month] = []
        by_month[month].append(image)

    # Show summary
    print(f"\nFound {len(results)} images:")
    for month, images in sorted(by_month.items()):
        print(f"  {month}: {len(images)} images")

    # Download
    print(f"\nDownloading to {outdir}...")
    paths = downloader.bulk_download(results, outdir)

    # Create metadata file
    metadata_file = os.path.join(outdir, "metadata.txt")
    with open(metadata_file, 'w') as f:
        f.write(f"Time Series Download\n")
        f.write(f"Tile: {tile_id}\n")
        f.write(f"Period: {start_date} to {end_date}\n")
        f.write(f"Total images: {len(results)}\n\n")

        for product_id, image in sorted(results.items(), key=lambda x: x[1].date):
            f.write(f"{image.date},{image.sensor},{image.filename}\n")

    print(f"✓ Metadata saved in {metadata_file}")
    return results, paths

# Use
download_time_series(
    tile_id="30TWM",
    start_date="2023-01-01",
    end_date="2023-12-31",
    outdir="./time_series_madrid"
)

    Example 2: Multi-Satellite Comparison¶

    Download data from same area with different satellites:

    from sat_download.api.odata import ODataAPI
from sat_download.api.usgs import USGSAPI
from sat_download.services.downloader import SatelliteImageDownloader
from sat_download.data_types.search import SearchFilters
from sat_download.enums import COLLECTIONS

def compare_satellites(lon, lat, date_start, date_end, output_base):
    """
    Download images from same area with Sentinel-2, Sentinel-3 and Landsat 8.
    """
    results_all = {}

    # Copernicus (Sentinel-2 and Sentinel-3)
    print("=== Searching in Copernicus ===")
    copernicus_api = ODataAPI(username="...", password="...")
    copernicus_dl = SatelliteImageDownloader(copernicus_api, verbose=1)

    geometry = f"POINT({lon} {lat})"

    # Sentinel-2
    print("\n1. Sentinel-2:")
    s2_filters = SearchFilters(
        collection=COLLECTIONS.SENTINEL_2.value,
        processing_level="L2A",
        start_date=date_start,
        end_date=date_end,
        geometry=geometry
    )
    s2_results = copernicus_dl.search(s2_filters)
    print(f"   Found: {len(s2_results)}")
    results_all['Sentinel-2'] = s2_results

    # Sentinel-3
    print("\n2. Sentinel-3:")
    s3_filters = SearchFilters(
        collection=COLLECTIONS.SENTINEL_3.value,
        start_date=date_start,
        end_date=date_end,
        geometry=geometry
    )
    s3_results = copernicus_dl.search(s3_filters)
    print(f"   Found: {len(s3_results)}")
    results_all['Sentinel-3'] = s3_results

    # USGS (Landsat 8)
    print("\n=== Searching in USGS ===")
    usgs_api = USGSAPI(username="...", password="...")
    usgs_dl = SatelliteImageDownloader(usgs_api, verbose=1)

    print("\n3. Landsat 8:")
    l8_filters = SearchFilters(
        collection=COLLECTIONS.LANDSAT_8.value,
        processing_level="T1",
        start_date=date_start,
        end_date=date_end,
        geometry=geometry
    )
    l8_results = usgs_dl.search(l8_filters)
    print(f"   Found: {len(l8_results)}")
    results_all['Landsat-8'] = l8_results

    # Download each type to its own directory
    print("\n=== Downloading ===")

    if s2_results:
        print("\nDownloading Sentinel-2...")
        copernicus_dl.bulk_download(s2_results, f"{output_base}/sentinel2")

    if s3_results:
        print("\nDownloading Sentinel-3...")
        copernicus_dl.bulk_download(s3_results, f"{output_base}/sentinel3")

    if l8_results:
        print("\nDownloading Landsat 8...")
        usgs_dl.bulk_download(l8_results, f"{output_base}/landsat8")

    # Create comparative report
    print("\n=== Comparative Report ===")
    for satellite, results in results_all.items():
        print(f"\n{satellite}:")
        print(f"  Total images: {len(results)}")
        if results:
            dates = [img.date for img in results.values()]
            print(f"  Period: {min(dates)} to {max(dates)}")

    return results_all

# Use - Barcelona area
results = compare_satellites(
    lon=2.1734,
    lat=41.3851,
    date_start="2024-06-01",
    date_end="2024-06-30",
    output_base="./comparison_barcelona"
)

    Example 3: Change Monitoring¶

    Detect changes by comparing two periods:

    from sat_download.api.odata import ODataAPI
from sat_download.services.downloader import SatelliteImageDownloader
from sat_download.data_types.search import SearchFilters
from sat_download.enums import COLLECTIONS
import os

def change_detection_download(tile_id, period_before, period_after, outdir):
    """
    Download images from two periods for change analysis.

    Parameters:
        tile_id (str): MGRS tile of interest
        period_before (tuple): (start_date, end_date) before
        period_after (tuple): (start_date, end_date) after
        outdir (str): Output directory
    """
    api = ODataAPI(username="...", password="...")
    downloader = SatelliteImageDownloader(api, verbose=1)

    # Helper function
    def download_period(period_name, start, end):
        filters = SearchFilters(
            collection=COLLECTIONS.SENTINEL_2.value,
            processing_level="L2A",
            start_date=start,
            end_date=end,
            tile_id=tile_id
        )

        print(f"\n=== Period {period_name}: {start} to {end} ===")
        results = downloader.search(filters)
        print(f"Found {len(results)} images")

        if results:
            # Select only one image (most recent)
            latest = max(results.items(), key=lambda x: x[1].date)
            selected = {latest[0]: latest[1]}

            period_dir = os.path.join(outdir, period_name.lower().replace(" ", "_"))
            print(f"Downloading: {latest[1].filename}")
            downloader.bulk_download(selected, period_dir)

            return selected
        return {}

    # Download both periods
    before_results = download_period("BEFORE", period_before[0], period_before[1])
    after_results = download_period("AFTER", period_after[0], period_after[1])

    # Create processing script
    script_path = os.path.join(outdir, "process_change.py")
    with open(script_path, 'w') as f:
        f.write("""#!/usr/bin/env python3
# Example script for change detection processing
import rasterio
import numpy as np

# Load images
# before = rasterio.open('./before/image.zip')
# after = rasterio.open('./after/image.zip')

# Calculate NDVI difference or other index
# ndvi_before = (nir_before - red_before) / (nir_before + red_before)
# ndvi_after = (nir_after - red_after) / (nir_after + red_after)
# change = ndvi_after - ndvi_before

print("Process images here")
""")

    print(f"\n✓ Example script created: {script_path}")

    return before_results, after_results

# Use - Monitor deforestation
before, after = change_detection_download(
    tile_id="20LLR",  # Amazon
    period_before=("2020-01-01", "2020-12-31"),
    period_after=("2023-01-01", "2023-12-31"),
    outdir="./change_detection"
)

    Example 4: Scheduled Download¶

    Automate periodic downloads:

    from sat_download.api.odata import ODataAPI
from sat_download.services.downloader import SatelliteImageDownloader
from sat_download.data_types.search import SearchFilters
from sat_download.enums import COLLECTIONS
from datetime import datetime, timedelta
import schedule
import time
import logging

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('downloads.log'),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)

class ScheduledDownloader:
    """Scheduled downloader for periodic updates."""

    def __init__(self, username, password, tile_id, output_dir):
        self.api = ODataAPI(username=username, password=password)
        self.downloader = SatelliteImageDownloader(self.api, verbose=1)
        self.tile_id = tile_id
        self.output_dir = output_dir
        self.last_download = None

    def download_latest(self):
        """Download most recent images."""
        logger.info(f"Starting scheduled download for tile {self.tile_id}")

        # Search last week
        end_date = datetime.now()
        start_date = end_date - timedelta(days=7)

        filters = SearchFilters(
            collection=COLLECTIONS.SENTINEL_2.value,
            processing_level="L2A",
            start_date=start_date.strftime("%Y-%m-%d"),
            end_date=end_date.strftime("%Y-%m-%d"),
            tile_id=self.tile_id
        )

        try:
            results = self.downloader.search(filters)

            if not results:
                logger.info("No new images found")
                return

            # Filter already downloaded images
            new_images = results
            if self.last_download:
                new_images = {
                    k: v for k, v in results.items()
                    if v.date > self.last_download
                }

            if not new_images:
                logger.info("No new images since last download")
                return

            logger.info(f"Found {len(new_images)} new images")

            # Download
            paths = self.downloader.bulk_download(new_images, self.output_dir)
            successful = len([p for p in paths if p])

            logger.info(f"Downloaded {successful}/{len(new_images)} images")

            # Update last datetime
            if new_images:
                latest_date = max(img.date for img in new_images.values())
                self.last_download = latest_date
                logger.info(f"Last downloaded image: {latest_date}")

        except Exception as e:
            logger.error(f"Error in scheduled download: {e}")

def main():
    """Start scheduler."""
    # Configuration
    COPERNICUS_USER = "your_email@example.com"
    COPERNICUS_PASS = "your_password"
    TILE_ID = "30TWM"
    OUTPUT_DIR = "./scheduled_downloads"

    # Create downloader
    scheduled = ScheduledDownloader(
        username=COPERNICUS_USER,
        password=COPERNICUS_PASS,
        tile_id=TILE_ID,
        output_dir=OUTPUT_DIR
    )

    # Schedule task
    schedule.every().monday.at("02:00").do(scheduled.download_latest)
    schedule.every().thursday.at("02:00").do(scheduled.download_latest)

    logger.info("Scheduler started")
    logger.info("Scheduled downloads: Monday and Thursday at 02:00")

    # Infinite loop
    try:
        while True:
            schedule.run_pending()
            time.sleep(60)  # Check every minute
    except KeyboardInterrupt:
        logger.info("Scheduler stopped by user")

if __name__ == "__main__":
    main()

    Example 5: Integration with Pandas¶

    Analyze metadata with Pandas:

    from sat_download.api.odata import ODataAPI
from sat_download.services.downloader import SatelliteImageDownloader
from sat_download.data_types.search import SearchFilters
from sat_download.enums import COLLECTIONS
import pandas as pd
from datetime import datetime

def create_metadata_dataframe(results):
    """Convert SearchResults to Pandas DataFrame."""
    data = []
    for product_id, image in results.items():
        data.append({
            'product_id': product_id,
            'date': datetime.strptime(image.date, '%Y%m%d'),
            'sensor': image.sensor,
            'satellite': image.identifier,
            'tile': image.tile,
            'filename': image.filename
        })

    df = pd.DataFrame(data)
    df = df.sort_values('date')
    return df

def analyze_coverage(tile_id, start_date, end_date):
    """Analyze temporal coverage of a tile."""
    # Search
    api = ODataAPI(username="...", password="...")
    downloader = SatelliteImageDownloader(api, verbose=0)

    filters = SearchFilters(
        collection=COLLECTIONS.SENTINEL_2.value,
        start_date=start_date,
        end_date=end_date,
        tile_id=tile_id
    )

    results = downloader.bulk_search(filters)

    if not results:
        print("No images found")
        return

    # Create DataFrame
    df = create_metadata_dataframe(results)

    # Statistical analysis
    print(f"\n=== Coverage Analysis: {tile_id} ===")
    print(f"Period: {start_date} to {end_date}")
    print(f"\nTotal images: {len(df)}")
    print(f"\nBy satellite:")
    print(df['satellite'].value_counts())

    # Monthly frequency
    print(f"\nImages per month:")
    monthly = df.set_index('date').resample('M').size()
    print(monthly)

    # Coverage gaps
    print(f"\nGap analysis:")
    df['days_since_last'] = df['date'].diff().dt.days
    max_gap = df['days_since_last'].max()
    print(f"Maximum gap: {max_gap:.0f} days")

    # Save report
    output_file = f"coverage_report_{tile_id}.csv"
    df.to_csv(output_file, index=False)
    print(f"\n✓ Report saved: {output_file}")

    return df

# Use
df = analyze_coverage(
    tile_id="30TWM",
    start_date="2023-01-01",
    end_date="2023-12-31"
)

# Additional analysis
print("\nQuarterly statistics:")
df_quarterly = df.set_index('date').resample('Q').agg({
    'product_id': 'count',
    'satellite': lambda x: x.value_counts().index[0] if len(x) > 0 else None
})
df_quarterly.columns = ['num_images', 'most_common_satellite']
print(df_quarterly)

    Example 6: Resilient Download with Retries¶

    Implement automatic retries:

    from sat_download.api.odata import ODataAPI
from sat_download.services.downloader import SatelliteImageDownloader
from sat_download.data_types.search import SearchFilters
import time
import logging

logger = logging.getLogger(__name__)

class ResilientDownloader:
    """Downloader with automatic retries."""

    def __init__(self, api, max_retries=3, retry_delay=60):
        self.downloader = SatelliteImageDownloader(api, verbose=1)
        self.max_retries = max_retries
        self.retry_delay = retry_delay

    def download_with_retry(self, product_id, image, outdir):
        """Download an image with retries."""
        for attempt in range(1, self.max_retries + 1):
            try:
                logger.info(f"Attempt {attempt}/{self.max_retries}: {image.filename}")

                path = self.downloader.download(
                    image_id=product_id,
                    out_dir=outdir,
                    outname=image.filename
                )

                if path:
                    logger.info(f"✓ Downloaded: {image.filename}")
                    return path
                else:
                    if attempt < self.max_retries:
                        logger.warning(f"Failed, retrying in {self.retry_delay}s...")
                        time.sleep(self.retry_delay)

            except Exception as e:
                logger.error(f"Error: {e}")
                if attempt < self.max_retries:
                    time.sleep(self.retry_delay)

        logger.error(f"✗ Failed after {self.max_retries} attempts: {image.filename}")
        return None

    def bulk_download_resilient(self, results, outdir):
        """Bulk download with retries."""
        paths = []

        for idx, (product_id, image) in enumerate(results.items(), 1):
            logger.info(f"\n[{idx}/{len(results)}] Processing {image.filename}")
            path = self.download_with_retry(product_id, image, outdir)
            paths.append(path)

        # Final report
        successful = len([p for p in paths if p])
        failed = len(paths) - successful

        logger.info(f"\n=== Final Report ===")
        logger.info(f"Successful: {successful}/{len(paths)}")
        logger.info(f"Failed: {failed}/{len(paths)}")

        return paths

# Use
api = ODataAPI(username="...", password="...")
resilient = ResilientDownloader(api, max_retries=3, retry_delay=60)

filters = SearchFilters(...)
results = api.search(filters)

paths = resilient.bulk_download_resilient(results, "./downloads")
